# 書き込むファイル

これは、最大5人で遊ぶ文化祭向けリアルタイムクイズアプリの完成形を、ファイルごとに集めた資料です。上から順に自分のプロジェクトへ作成・貼り付けしてください。

機能は次を含みます。

- 参加者のニックネーム登録・待機・2択／4択・早押し（PUSHまたはSpace）
- 大画面の待機ランキングTOP3、問題、残り時間、得点、結果、NEW RECORD
- 司会画面の進行、正解発表、早押しの正誤判定、音声再生、得点調整
- Django Adminでの問題登録、ゲームごとのランダム出題と問題の重複防止、ランキング保存

> まずは教室内など、信頼できるネットワークで使う前提のサンプルです。本番公開する場合は、司会画面にログイン認証を必ず追加してください。

## 2. 最初の準備

空のフォルダで、次を実行します。

```bash
python3 -m venv .venv
source .venv/bin/activate
pip install -r requirements.txt
django-admin startproject config .
python manage.py startapp quiz
```

`startproject` と `startapp` が同名のひな形を作るので、以下のコードで上書きします。最後にデータベースを作成します。

```bash
python manage.py makemigrations
python manage.py migrate
python manage.py createsuperuser
python manage.py runserver 0.0.0.0:8000
```

同じWi-Fiの端末からは、司会PCのIPアドレスを使って `http://192.168.x.x:8000/` のように開きます。

```python
import uuid

from django.core.exceptions import ValidationError
from django.db import models


class Question(models.Model):
    """Adminで登録する問題。選択式・早押し・音当てに共通で使う。"""

    class QuestionType(models.TextChoices):
        CHOICE = "choice", "2択・4択"
        BUZZER = "buzzer", "早押し"
        AUDIO = "audio", "音当て"

    title = models.CharField("管理用タイトル", max_length=100)
    question_type = models.CharField(
        "問題形式", max_length=10, choices=QuestionType.choices, default=QuestionType.CHOICE
    )
    category = models.CharField("カテゴリ", max_length=50, blank=True)
    body = models.TextField("問題文")
    choice_1 = models.CharField("選択肢1", max_length=200, blank=True)
    choice_2 = models.CharField("選択肢2", max_length=200, blank=True)
    choice_3 = models.CharField("選択肢3", max_length=200, blank=True)
    choice_4 = models.CharField("選択肢4", max_length=200, blank=True)
    correct_choice = models.PositiveSmallIntegerField("正解の番号", null=True, blank=True)
    correct_text = models.CharField("正解表示・口頭回答の正解", max_length=300, blank=True)
    explanation = models.TextField("解説", blank=True)
    image = models.ImageField("問題画像", upload_to="questions/images/", blank=True)
    audio_file = models.FileField("音声ファイル", upload_to="questions/audio/", blank=True)
    points = models.PositiveIntegerField("正解時の得点", default=10)
    is_active = models.BooleanField("出題に使う", default=True)
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ["-created_at"]

    def __str__(self):
        return f"[{self.get_question_type_display()}] {self.title}"

    @property
    def choices(self):
        return [value for value in [self.choice_1, self.choice_2, self.choice_3, self.choice_4] if value]

    def clean(self):
        if self.question_type == self.QuestionType.CHOICE:
            if len(self.choices) not in (2, 4):
                raise ValidationError("選択式では選択肢を2個または4個入力してください。")
            if not self.correct_choice or self.correct_choice > len(self.choices):
                raise ValidationError("正解の番号を選択肢の範囲内で入力してください。")
        if self.question_type == self.QuestionType.AUDIO and not self.audio_file:
            raise ValidationError("音当て問題には音声ファイルを登録してください。")


class Game(models.Model):
    class Status(models.TextChoices):
        LOBBY = "lobby", "待機中"
        RUNNING = "running", "進行中"
        FINISHED = "finished", "終了"

    class Phase(models.TextChoices):
        LOBBY = "lobby", "待機"
        QUESTION = "question", "問題・回答中"
        ANSWER_CLOSED = "answer_closed", "回答締切"
        BUZZER_OPEN = "buzzer_open", "早押し受付中"
        BUZZER_LOCKED = "buzzer_locked", "司会判定中"
        RESULT = "result", "結果発表"
        FINISHED = "finished", "終了"

    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    title = models.CharField(max_length=100, default="文化祭クイズ")
    max_players = models.PositiveSmallIntegerField(default=5)
    question_count = models.PositiveSmallIntegerField(default=10)
    seconds_per_question = models.PositiveSmallIntegerField(default=20)
    status = models.CharField(max_length=10, choices=Status.choices, default=Status.LOBBY)
    phase = models.CharField(max_length=20, choices=Phase.choices, default=Phase.LOBBY)
    current_index = models.IntegerField(default=-1)
    question_started_at = models.DateTimeField(null=True, blank=True)
    buzzed_player = models.ForeignKey(
        "GamePlayer", null=True, blank=True, on_delete=models.SET_NULL, related_name="current_buzz_games"
    )
    created_at = models.DateTimeField(auto_now_add=True)

    def __str__(self):
        return f"{self.title} ({str(self.id)[:8]})"


class GameQuestion(models.Model):
    """このゲームでの出題順。Questionをランダムに一度だけ並べる。"""

    game = models.ForeignKey(Game, on_delete=models.CASCADE, related_name="game_questions")
    question = models.ForeignKey(Question, on_delete=models.PROTECT)
    position = models.PositiveSmallIntegerField()

    class Meta:
        constraints = [
            models.UniqueConstraint(fields=["game", "position"], name="unique_game_position"),
            models.UniqueConstraint(fields=["game", "question"], name="no_duplicate_question_in_game"),
        ]
        ordering = ["position"]


class GamePlayer(models.Model):
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    game = models.ForeignKey(Game, on_delete=models.CASCADE, related_name="players")
    nickname = models.CharField(max_length=20)
    slot = models.PositiveSmallIntegerField()
    score = models.IntegerField(default=0)
    connected = models.BooleanField(default=True)
    joined_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        constraints = [
            models.UniqueConstraint(fields=["game", "nickname"], name="unique_nickname_in_game"),
            models.UniqueConstraint(fields=["game", "slot"], name="unique_slot_in_game"),
        ]
        ordering = ["slot"]

    def __str__(self):
        return self.nickname


class PlayerAnswer(models.Model):
    game_question = models.ForeignKey(GameQuestion, on_delete=models.CASCADE, related_name="answers")
    player = models.ForeignKey(GamePlayer, on_delete=models.CASCADE, related_name="answers")
    selected_choice = models.PositiveSmallIntegerField()
    is_correct = models.BooleanField()
    answered_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        constraints = [
            models.UniqueConstraint(fields=["game_question", "player"], name="one_choice_answer_per_player"),
        ]


class BuzzerAttempt(models.Model):
    game_question = models.ForeignKey(GameQuestion, on_delete=models.CASCADE, related_name="buzz_attempts")
    player = models.ForeignKey(GamePlayer, on_delete=models.CASCADE, related_name="buzz_attempts")
    is_correct = models.BooleanField(null=True, blank=True)
    score_delta = models.IntegerField(default=0)
    buzzed_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        # 不正解だった参加者は、その問題で再び早押しできない。
        constraints = [
            models.UniqueConstraint(fields=["game_question", "player"], name="one_buzz_per_player"),
        ]
        ordering = ["buzzed_at"]


class RankingEntry(models.Model):
    """ゲーム終了時点の順位を保存するためのスナップショット。"""

    game = models.ForeignKey(Game, on_delete=models.CASCADE, related_name="ranking_entries")
    player_name = models.CharField(max_length=20)
    score = models.IntegerField()
    rank = models.PositiveSmallIntegerField()
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ["rank"]
```

```python
from django.http import Http404
from django.shortcuts import get_object_or_404, redirect, render
from django.views.decorators.http import require_POST

from .models import Game


def home(request):
    return render(request, "quiz/home.html")


@require_POST
def new_game(request):
    title = request.POST.get("title", "文化祭クイズ").strip()[:100] or "文化祭クイズ"
    try:
        question_count = max(1, min(int(request.POST.get("question_count", 10)), 100))
        seconds = max(5, min(int(request.POST.get("seconds", 20)), 120))
    except ValueError:
        question_count, seconds = 10, 20

    game = Game.objects.create(
        title=title,
        question_count=question_count,
        seconds_per_question=seconds,
    )
    return redirect("quiz:control", game_id=game.id)


def game_or_404(game_id):
    return get_object_or_404(Game, id=game_id)


def control(request, game_id):
    return render(request, "quiz/control.html", {"game": game_or_404(game_id)})


def display(request, game_id):
    return render(request, "quiz/display.html", {"game": game_or_404(game_id)})

    return render(request, "quiz/player.html", {"game": game_or_404(game_id)})
```

### `quiz/consumers.py`

```python
from datetime import timedelta

from channels.db import database_sync_to_async
from channels.generic.websocket import AsyncJsonWebsocketConsumer
from django.db import IntegrityError, transaction
from django.db.models import F, Max
from django.utils import timezone

from .models import BuzzerAttempt, Game, GamePlayer, GameQuestion, PlayerAnswer, Question, RankingEntry


class QuizConsumer(AsyncJsonWebsocketConsumer):
    """全画面をゲームIDごとのグループへ接続し、DBを正しい状態の唯一の基準にする。"""

    async def connect(self):
        self.game_id = self.scope["url_route"]["kwargs"]["game_id"]
        self.group_name = f"quiz_{self.game_id}"
        self.player_id = None
        if not await self.game_exists():
            await self.close(code=4404)
            return
        await self.channel_layer.group_add(self.group_name, self.channel_name)
        await self.accept()
        await self.send_state()

    async def disconnect(self, close_code):
        if self.player_id:
            await self.set_connected(self.player_id, False)
            await self.broadcast_state()
        await self.channel_layer.group_discard(self.group_name, self.channel_name)

    async def receive_json(self, content, **kwargs):
        action = content.get("action")
        try:
            if action == "request_state":
                await self.send_state()
                return

            if action == "join_player":
                player = await self.join_player(content.get("nickname", ""), content.get("player_id"))
                self.player_id = str(player["id"])
                await self.send_json({"type": "joined", "player_id": self.player_id, "nickname": player["nickname"]})
                await self.broadcast_state()
                return

            if action == "submit_answer":
                await self.submit_answer(int(content.get("choice")))
            elif action == "buzz":
                await self.buzz()
            elif action == "start_next_question":
                await self.start_next_question()
            elif action == "close_answers":
                await self.close_answers()
            elif action == "reveal_result":
                await self.reveal_result()
            elif action == "open_buzzer":
                await self.open_buzzer()
            elif action == "judge_buzzer":
                await self.judge_buzzer(bool(content.get("correct")))
            elif action == "adjust_score":
                await self.adjust_score(content.get("player_id"), int(content.get("delta")))
            elif action == "set_seconds":
                await self.set_seconds(int(content.get("seconds")))
            elif action == "play_audio":
                await self.channel_layer.group_send(self.group_name, {"type": "audio.play"})
                return
            else:
                raise ValueError("不明な操作です。")
            await self.broadcast_state()
        except (ValueError, TypeError, IntegrityError) as error:
            await self.send_json({"type": "error", "message": str(error) or "操作できません。"})

    async def state_update(self, event):
        await self.send_state()

    async def audio_play(self, event):
        await self.send_json({"type": "play_audio"})

    async def broadcast_state(self):
        await self.channel_layer.group_send(self.group_name, {"type": "state.update"})

    async def send_state(self):
        await self.send_json({"type": "state", "state": await self.get_state()})

    @database_sync_to_async
    def game_exists(self):
        return Game.objects.filter(id=self.game_id).exists()

    @database_sync_to_async
    def set_connected(self, player_id, connected):
        GamePlayer.objects.filter(id=player_id, game_id=self.game_id).update(connected=connected)

    @database_sync_to_async
    def join_player(self, raw_nickname, old_player_id):
        nickname = str(raw_nickname).strip()[:20]
        if not nickname:
            raise ValueError("ニックネームを入力してください。")

        with transaction.atomic():
            game = Game.objects.select_for_update().get(id=self.game_id)
            if game.status == Game.Status.FINISHED:
                raise ValueError("このゲームは終了しています。")

            # ページ再読み込み時は、ブラウザに保存したIDで同じ参加者に戻る。
            if old_player_id:
                player = GamePlayer.objects.filter(id=old_player_id, game=game).first()
                if player:
                    player.connected = True
                    player.save(update_fields=["connected"])
                    return {"id": player.id, "nickname": player.nickname}

            if game.players.count() >= game.max_players:
                raise ValueError("参加者は最大5人です。")
            if game.players.filter(nickname=nickname).exists():
                raise ValueError("そのニックネームはすでに使われています。")

            used_slots = set(game.players.values_list("slot", flat=True))
            slot = next(number for number in range(1, game.max_players + 1) if number not in used_slots)
            player = GamePlayer.objects.create(game=game, nickname=nickname, slot=slot)
            return {"id": player.id, "nickname": player.nickname}

    @database_sync_to_async
    def submit_answer(self, choice):
        if not self.player_id:
            raise ValueError("先に参加してください。")
        with transaction.atomic():
            game, game_question = self.lock_current_game_question()
            if game.phase != Game.Phase.QUESTION or game_question.question.question_type != Question.QuestionType.CHOICE:
                raise ValueError("現在は選択式の回答時間ではありません。")
            if choice not in range(1, len(game_question.question.choices) + 1):
                raise ValueError("選択肢が正しくありません。")
            player = GamePlayer.objects.select_for_update().get(id=self.player_id, game=game)
            if PlayerAnswer.objects.filter(game_question=game_question, player=player).exists():
                raise ValueError("この問題にはすでに回答済みです。")
            correct = choice == game_question.question.correct_choice
            PlayerAnswer.objects.create(game_question=game_question, player=player, selected_choice=choice, is_correct=correct)
            if correct:
                player.score = F("score") + game_question.question.points
                player.save(update_fields=["score"])

    @database_sync_to_async
    def buzz(self):
        if not self.player_id:
            raise ValueError("先に参加してください。")
        with transaction.atomic():
            game, game_question = self.lock_current_game_question()
            if game.phase != Game.Phase.BUZZER_OPEN:
                raise ValueError("現在は早押しを受け付けていません。")
            player = GamePlayer.objects.get(id=self.player_id, game=game)
            if BuzzerAttempt.objects.filter(game_question=game_question, player=player).exists():
                raise ValueError("この問題ではすでに早押し済みです。")
            BuzzerAttempt.objects.create(game_question=game_question, player=player)
            game.buzzed_player = player
            game.phase = Game.Phase.BUZZER_LOCKED
            game.save(update_fields=["buzzed_player", "phase"])

    @database_sync_to_async
    def start_next_question(self):
        with transaction.atomic():
            game = Game.objects.select_for_update().get(id=self.game_id)
            if game.status == Game.Status.LOBBY:
                self.create_random_questions(game)
                game.status = Game.Status.RUNNING

            next_index = game.current_index + 1
            if not GameQuestion.objects.filter(game=game, position=next_index).exists():
                self.finish_game(game)
                return

            question = GameQuestion.objects.select_related("question").get(game=game, position=next_index).question
            game.current_index = next_index
            game.buzzed_player = None
            game.question_started_at = timezone.now()
            game.phase = Game.Phase.QUESTION if question.question_type == Question.QuestionType.CHOICE else Game.Phase.BUZZER_OPEN
            game.save(update_fields=["status", "current_index", "buzzed_player", "question_started_at", "phase"])

    @database_sync_to_async
    def close_answers(self):
        game, game_question = self.lock_current_game_question()
        if game.phase != Game.Phase.QUESTION:
            raise ValueError("選択式の回答時間ではありません。")
        game.phase = Game.Phase.ANSWER_CLOSED
        game.save(update_fields=["phase"])

    @database_sync_to_async
    def reveal_result(self):
        game, _ = self.lock_current_game_question()
        if game.phase not in (Game.Phase.QUESTION, Game.Phase.ANSWER_CLOSED, Game.Phase.BUZZER_LOCKED):
            raise ValueError("現在は結果を発表できません。")
        game.phase = Game.Phase.RESULT
        game.save(update_fields=["phase"])

    @database_sync_to_async
    def open_buzzer(self):
        game, game_question = self.lock_current_game_question()
        if game_question.question.question_type not in (Question.QuestionType.BUZZER, Question.QuestionType.AUDIO):
            raise ValueError("早押し問題ではありません。")
        game.buzzed_player = None
        game.phase = Game.Phase.BUZZER_OPEN
        game.save(update_fields=["buzzed_player", "phase"])

    @database_sync_to_async
    def judge_buzzer(self, correct):
        with transaction.atomic():
            game, game_question = self.lock_current_game_question()
            if game.phase != Game.Phase.BUZZER_LOCKED or not game.buzzed_player_id:
                raise ValueError("判定する早押し回答者がいません。")
            attempt = BuzzerAttempt.objects.select_for_update().get(game_question=game_question, player=game.buzzed_player)
            delta = game_question.question.points if correct else 0
            attempt.is_correct = correct
            attempt.score_delta = delta
            attempt.save(update_fields=["is_correct", "score_delta"])
            if delta:
                GamePlayer.objects.filter(id=game.buzzed_player_id).update(score=F("score") + delta)
            game.buzzed_player = None
            # 不正解なら次の人が押せる。正解なら結果発表へ。
            game.phase = Game.Phase.RESULT if correct else Game.Phase.BUZZER_OPEN
            game.save(update_fields=["buzzed_player", "phase"])

    @database_sync_to_async
    def adjust_score(self, player_id, delta):
        delta = max(-100, min(delta, 100))
        if not player_id:
            raise ValueError("参加者を選んでください。")
        updated = GamePlayer.objects.filter(id=player_id, game_id=self.game_id).update(score=F("score") + delta)
        if not updated:
            raise ValueError("参加者が見つかりません。")

    @database_sync_to_async
    def set_seconds(self, seconds):
        seconds = max(5, min(seconds, 120))
        Game.objects.filter(id=self.game_id, status=Game.Status.LOBBY).update(seconds_per_question=seconds)

    def lock_current_game_question(self):
        game = Game.objects.select_for_update().select_related("buzzed_player").get(id=self.game_id)
        game_question = GameQuestion.objects.select_related("question").get(game=game, position=game.current_index)
        return game, game_question

    def create_random_questions(self, game):
        if game.game_questions.exists():
            return
        candidates = list(Question.objects.filter(is_active=True).order_by("?")[:game.question_count])
        if not candidates:
            raise ValueError("Adminで有効な問題を1問以上登録してください。")
        GameQuestion.objects.bulk_create([
            GameQuestion(game=game, question=question, position=index)
            for index, question in enumerate(candidates)
        ])

    def finish_game(self, game):
        game.status = Game.Status.FINISHED
        game.phase = Game.Phase.FINISHED
        game.buzzed_player = None
        game.save(update_fields=["status", "phase", "buzzed_player"])
        RankingEntry.objects.filter(game=game).delete()
        players = list(game.players.order_by("-score", "joined_at"))
        rankings, previous_score, previous_rank = [], None, 0
        for position, player in enumerate(players, start=1):
            rank = previous_rank if player.score == previous_score else position
            rankings.append(RankingEntry(game=game, player_name=player.nickname, score=player.score, rank=rank))
            previous_score, previous_rank = player.score, rank
        RankingEntry.objects.bulk_create(rankings)

    @database_sync_to_async
    def get_state(self):
        game = Game.objects.select_related("buzzed_player").get(id=self.game_id)
        players = list(game.players.order_by("-score", "slot"))
        game_questions = list(GameQuestion.objects.filter(game=game).select_related("question").order_by("position"))
        current_gq = next((item for item in game_questions if item.position == game.current_index), None)

        question_data = None
        answer_data = []
        if current_gq:
            question = current_gq.question
            question_data = {
                "number": game.current_index + 1,
                "total": len(game_questions),
                "type": question.question_type,
                "category": question.category,
                "body": question.body,
                "choices": question.choices,
                "points": question.points,
                "image_url": question.image.url if question.image else "",
                "audio_url": question.audio_file.url if question.audio_file else "",
            }
            # 正解は結果発表以降だけ送る。回答中の画面には送らない。
            if game.phase in (Game.Phase.RESULT, Game.Phase.FINISHED):
                question_data.update({
                    "correct_choice": question.correct_choice,
                    "correct_text": question.correct_text or (question.choices[question.correct_choice - 1] if question.correct_choice else ""),
                    "explanation": question.explanation,
                })
                answer_data = list(
                    current_gq.answers.select_related("player").values(
                        "player__nickname", "selected_choice", "is_correct"
                    )
                )

        answered_ids = []
        if current_gq:
            answered_ids = list(current_gq.answers.values_list("player_id", flat=True))

        rankings = list(game.ranking_entries.values("rank", "player_name", "score"))
        previous_best = RankingEntry.objects.exclude(game=game).aggregate(best=Max("score"))["best"]
        current_best = rankings[0]["score"] if rankings else None
        return {
            "game": {
                "id": str(game.id), "title": game.title, "status": game.status, "phase": game.phase,
                "seconds": game.seconds_per_question,
                "deadline": (game.question_started_at + timedelta(seconds=game.seconds_per_question)).isoformat() if game.question_started_at else None,
                "buzzed_player_id": str(game.buzzed_player_id) if game.buzzed_player_id else None,
            },
            "question": question_data,
            "players": [
                {"id": str(p.id), "nickname": p.nickname, "slot": p.slot, "score": p.score, "connected": p.connected,
                 "answered": str(p.id) in {str(answered) for answered in answered_ids}}
                for p in players
            ],
            "answers": answer_data,
            "rankings": rankings,
            "is_new_record": current_best is not None and (previous_best is None or current_best > previous_best),
        }
```

```javascript
function createQuizSocket(gameId, onState, onEvent) {
  const protocol = location.protocol === "https:" ? "wss" : "ws";
  const socket = new WebSocket(
    `${protocol}://${location.host}/ws/quiz/${gameId}/`,
  );
  socket.addEventListener("message", (event) => {
    const message = JSON.parse(event.data);
    if (message.type === "state") onState(message.state);
    else onEvent?.(message);
  });
  socket.addEventListener("close", () =>
    onEvent?.({
      type: "error",
      message: "通信が切れました。画面を再読み込みしてください。",
    }),
  );
  return {
    send(action, extra = {}) {
      if (socket.readyState === WebSocket.OPEN)
        socket.send(JSON.stringify({ action, ...extra }));
    },
  };
}

function escapeHtml(value) {
  const div = document.createElement("div");
  div.textContent = value ?? "";
  return div.innerHTML;
}
```
